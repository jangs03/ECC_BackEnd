   -> AS avgnull0 FROM sample51;

# 예제 5-15. 집계한 결과 정렬하기

   -> FROM sample51 GROUP BY name ORDER BY SUM(quantity) DESC;

# 예제 5-32. NOT EXISTS로 '없음'으로 갱신하기

# 예제 5-33. 열에 테이블명 붙이기

   -> (SELECT no2 FROM sample552);
